async function importBackup(e) { const f = e.target.files[0]; if (!f) return; const r = new FileReader(); r.onload = async ev => { try { const data = JSON.parse(ev.target.result); if (!confirm('سيتم استبدال جميع البيانات. متابعة؟')) return; showToast('⏳ جاري الاستيراد...'); state.managers = data.managers || []; state.vacancies = data.vacancies || []; state.candidates = data.candidates || []; state.interviews = data.interviews || []; if (data.language) state.language = data.language; if (!state.managers.some(m => m.username === ADMIN_USERNAME)) { state.managers.unshift({ id: uid(), name: 'مدير النظام', username: ADMIN_USERNAME, password: ADMIN_PASSWORD, role: 'admin', permissions: { canViewAllVacancies:true, canAddVacancy:true, canViewCV:true, canAddCV:true, allowedVacancyIds:[] } }); } // ✅ رفع كل CV إلى المسار المنفصل let cvCount = 0; for (const c of state.candidates) { if (c.cvData && c.cvData.length > 100) { try { await saveCV(c.id, c.cvData, c.cvName, c.cvType); c.hasCV = true; cvCount++; } catch (err) { console.warn('فشل رفع CV لـ', c.name, err); } } } await saveAll(); showToast(`✅ تم الاستيراد! (${cvCount} CV رُفعت)`); render(); } catch (err) { console.error(err); showToast('ملف غير صالح', 'error'); } }; r.readAsText(f); }
